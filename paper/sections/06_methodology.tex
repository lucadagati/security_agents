\section{Experimental Methodology}
\label{sec:method}

\subsection{Metrics}
Let $\mathcal{E}_T$ be the set of episodes in a run.
\begin{align}
  \mathrm{ASR}
  &=
  \frac{\bigl|\{\,e\in\mathcal{E}_T : \text{exfiltration succeeded}\,\}\bigr|}
       {\bigl|\{\,e\in\mathcal{E}_T : \text{attack attempted}\,\}\bigr|},
  \label{eq:asr}\\[2pt]
  \mathrm{DR}
  &=
  \frac{\bigl|\{\,e\in\mathcal{E}_T : \text{at least one detection}\,\}\bigr|}
       {\bigl|\{\,e\in\mathcal{E}_T : \text{attack attempted}\,\}\bigr|}.
  \label{eq:dr}
\end{align}
Attack and detection costs are mean resource consumption. Recovery time
is the mean number of turns from first compromise to restored isolation
of the data store. Coalition rate is the fraction of episodes in which
a reciprocal same-role component of size $\ge 2$ appears. Deception
rate is the fraction of actions that are decoy deployments or
stealth-pacing waits taken while holding a foothold.

\paragraph{Adaptation gain.}
$\mathrm{AG}=\mathrm{Perf}_{\mathrm{adaptive}}-\mathrm{Perf}_{\mathrm{static}}$.

\paragraph{Strategy diversity and novelty.}
Diversity is mean pairwise signature distance. Novelty is the fraction
of episodes that introduce a taxonomy tag not seen before.

\paragraph{Co-evolutionary pressure.}
With signatures $s_A^{(t)},s_D^{(t)}$ of the two populations,
\begin{equation}
  \mathrm{CEP}
  \;=\;
  \frac{1}{T-1}
  \sum_{t=1}^{T-1}
  \frac{\bigl\|s_A^{(t+1)}-s_A^{(t)}\bigr\|_2
        +\bigl\|s_D^{(t+1)}-s_D^{(t)}\bigr\|_2}{2}.
  \label{eq:cep}
\end{equation}
High CEP indicates an active arms race (RQ5).

\paragraph{Emergent security risk.}
\begin{equation}
  \mathrm{ESR}
  \;=\;
  \frac{P(\mathrm{failure}\mid \mathrm{interaction})}
       {P(\mathrm{failure}\mid \mathrm{isolation})}.
  \label{eq:esr}
\end{equation}
Values above $1$ mean that interaction itself introduces risk. ESR
requires a paired run with communication disabled.

\subsection{Baselines}
\begin{itemize}
\item B1/B2: rule-based attacker/defender (L0).
\item B3: LLM agent, null memory (static).
\item B4: LLM agent, episodic memory.
\item B5: LLM agent, persistent memory.
\item Heuristic adaptive policies with a stealth/vigilance parameter,
used to validate the loop without a language model.
\end{itemize}

\subsection{Hypotheses}
\begin{itemize}
\item \textbf{H1} $\mathrm{ASR}_{\mathrm{adaptive}}>\mathrm{ASR}_{\mathrm{static}}$.
\item \textbf{H2} Adaptive defenders suppress ASR, after which attackers
recover by changing strategy.
\item \textbf{H3} ESR increases with interaction density.
\item \textbf{H4} Persistent memory produces stronger adaptation than
episodic memory.
\item \textbf{H5} Some behaviours appear that are not encoded in
objectives.
\item \textbf{H6} Population size changes qualitative behaviour
(coalitions, collective strategies).
\item \textbf{H7} Higher model capability need not produce safer
behaviour; it may raise both offensive and defensive sophistication.
\end{itemize}

\subsection{Reproducibility}
Every run is a YAML file specifying models, counts, topology, memory,
communication, seed, episode count, backend and budget. The ladder in
Section~\ref{sec:results} uses seed $100$, $40$ episodes, $20$ turns,
three service hosts and two vulnerabilities per host, on the simulator.
